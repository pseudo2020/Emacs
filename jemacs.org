* Architecture
This is the literate source for the JEmacs configuration.

The generated configuration is deliberately layered:

- =early-init.el=: startup-time settings that must happen before the normal
  package/frame initialization.
- =init.el=: paths and entry point only.
- =config.el=: top-level module loader and post-startup cleanup.
- =lisp/jemacs-core.el=: package setup and global Emacs defaults.
- =lisp/jemacs-ui.el=: frames, faces, and visual presentation.
- =lisp/jemacs-editing.el=: editing behavior shared across modes.
- =lisp/jemacs-completion.el=: minibuffer and in-buffer completion.
- =lisp/jemacs-programming.el=: programming-wide tooling such as Eglot.
- =lisp/jemacs-lang-*.el=: language-specific configuration.
- =lisp/jemacs-projects.el=: project management.
- =lisp/jemacs-terminal.el=: terminal integration.
- =lisp/jemacs-org.el=: Org configuration.
- =lisp/jemacs-git.el=: Git tooling.
- =lisp/jemacs-dashboard.el=: startup dashboard.
- =lisp/jemacs-llm.el=: LLM integration.
- =lisp/jemacs-custom.el=: personal functions and small one-off behavior.

Each module should have one responsibility. If a package is only useful for
one domain, configure it in that domain's module rather than in =core=.

* Bootstrap

** Early initialization
#+begin_src emacs-lisp :tangle early-init.el
(setq gc-cons-threshold (* 100 1024 1024)
      gc-cons-percentage 0.6)
#+end_src

** Init
Keep =init.el= intentionally small: it establishes paths and delegates all
configuration to =config.el=.

#+begin_src emacs-lisp :tangle init.el
(defconst jemacs-config-dir
  (file-name-directory (or load-file-name buffer-file-name))
  "Directory containing the JEmacs configuration.")

(defconst jemacs-config-org
  (expand-file-name "jemacs.org" jemacs-config-dir)
  "Literate source for the JEmacs configuration.")

(defconst jemacs-config-lisp-dir
  (expand-file-name "lisp/" jemacs-config-dir)
  "Directory containing generated JEmacs modules.")

(add-to-list 'load-path jemacs-config-lisp-dir)

(require 'config)
#+end_src

** Top-level loader
The loader is the only place that decides module load order.

#+begin_src emacs-lisp :tangle config.el
(require 'jemacs-core)
(require 'jemacs-ui)
(require 'jemacs-editing)
(require 'jemacs-completion)
(require 'jemacs-programming)
(require 'jemacs-lang-c)

(require 'jemacs-lang-lua)
(require 'jemacs-lang-lisp)
(require 'jemacs-projects)
(require 'jemacs-terminal)
(require 'jemacs-org)
(require 'jemacs-git)
(require 'jemacs-dashboard)
(require 'jemacs-llm)
(require 'jemacs-custom)

(add-hook
 'emacs-startup-hook
 (lambda ()
   (setq gc-cons-threshold (* 16 1024 1024)
         gc-cons-percentage 0.1)))
#+end_src

* Core

Global package management and Emacs-wide defaults belong here. Nothing
domain-specific should be added to this module.

#+begin_src emacs-lisp :tangle lisp/jemacs-core.el
(require 'package)

(when (native-comp-available-p)
  (setq native-comp-async-report-warnings-errors nil))

(setq package-archives
      '(("gnu"    . "https://elpa.gnu.org/packages/")
        ("nongnu" . "https://elpa.nongnu.org/nongnu/")
        ("melpa"  . "https://melpa.org/packages/")))

(require 'use-package)
(setq use-package-always-ensure t)

(use-package emacs
  :ensure nil
  :custom
  (indent-tabs-mode nil)
  (tab-width 2)
  (standard-indent 2)
  (fill-column 80)
  (sentence-end-double-space nil)
  (truncate-lines t)
  (inhibit-startup-screen t)
  (initial-scratch-message nil)
  (ring-bell-function #'ignore)
  (visible-bell nil)
  (make-backup-files t)
  (backup-directory-alist
   `(("." . ,(expand-file-name "backups/" user-emacs-directory))))
  (auto-save-default t)
  (auto-save-file-name-transforms
   `((".*" ,(expand-file-name "auto-save/" user-emacs-directory) t)))
  (create-lockfiles t)
  (savehist-file
   (expand-file-name "etc/savehist" user-emacs-directory))
  (save-place-file
   (expand-file-name "etc/save-place" user-emacs-directory))
  (recentf-save-file
   (expand-file-name "etc/recentf" user-emacs-directory))
  :init
  (fset 'yes-or-no-p #'y-or-n-p))

(use-package which-key
  :ensure nil
  :config
  (which-key-mode 1))
#+end_src

* Completion

All minibuffer and in-buffer completion packages belong together so their
configuration can be changed without touching unrelated modules.

#+begin_src emacs-lisp :tangle lisp/jemacs-completion.el
(use-package vertico
  :init
  (vertico-mode 1))

(use-package orderless
  :custom
  (completion-styles '(orderless basic)))

(use-package marginalia
  :init
  (marginalia-mode 1))

(use-package consult
  :bind
  (("C-'"   . consult-line)
   ("C-x b" . consult-buffer)
   ("M-y"   . consult-yank-pop)
   ("C-c h" . consult-history)
   ("C-c k" . consult-ripgrep)))

(use-package corfu
  :custom
  (corfu-auto t)
  (corfu-auto-delay 0.1)
  (corfu-auto-prefix 2)
  :init
  (global-corfu-mode 1))
#+end_src

* Programming

** Shared programming infrastructure
Only cross-language tooling belongs here.

#+begin_src emacs-lisp :tangle lisp/jemacs-programming.el
(defun jemacs-prog-font-scale ()
  "Apply the preferred font scale to programming buffers."
  (text-scale-set 1))

(add-hook 'prog-mode-hook #'jemacs-prog-font-scale)
(add-hook 'prog-mode-hook #'display-line-numbers-mode)

(use-package eglot
  :ensure nil
  :bind
  (:map eglot-mode-map
        ("C-c f" . eglot-format-buffer))
  :config
  (add-to-list
   'eglot-server-programs
   '(c-mode c++-mode . ("clangd")))
  (add-to-list
   'eglot-server-programs
   '(python-mode . ("pyright-langserver" "--stdio"))))
#+end_src
